/**
 * 管理画面用の初期データ取得
 * @return {Object}
 */
function getAdminInitialData() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const webAppUrl = getWebAppUrl_();
    const staffList = loadStaffList_(ss, false);
    const items = loadItems_(ss, { includeInactive: true });
    const reasons = loadReasons_(ss, { includeInactive: true });

    return {
      success: true,
      data: {
        businessTypes: CONFIG.BUSINESS_TYPES,
        themes: CONFIG.THEMES,
        items: items,
        reasons: reasons, // { [businessType]: { [itemKey]: Array<Reason> } }
        staffList: staffList,
        webAppUrl: webAppUrl,
        reservedKeys: CONFIG.RESERVED_KEYS
      }
    };
  } catch (error) {
    console.error('getAdminInitialData エラー:', error);
    return {
      success: false,
      message: 'マスタデータの読み込み中にエラーが発生しました: ' + error.message
    };
  }
}

/**
 * 項目マスタの保存および回答シートのヘッダー自動同期
 * @param {string} businessType 
 * @param {Array<Object>} items 
 * @return {Object}
 */
function saveItemMaster(businessType, items) {
  return withLock_(() => {
    if (!businessType || !Array.isArray(items)) {
      return { success: false, message: '不正なパラメータです。' };
    }

    // 予約キーとの衝突チェック
    const reserved = new Set(CONFIG.RESERVED_KEYS);
    for (const item of items) {
      const key = String(item.key || '').trim();
      if (reserved.has(key)) {
        return { success: false, message: `項目キー「${key}」はシステム予約語のため使用できません。別のキーを指定してください。` };
      }
    }

    // 整骨院の場合は金額トリガーを強制的にOFFに補正
    if (businessType === '整骨院') {
      items.forEach(item => {
        if (Array.isArray(item.options)) {
          item.options.forEach(opt => {
            opt.isAmountTrigger = false;
          });
        }
      });
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    saveItems_(ss, businessType, items);
    syncAnswerSheetHeaders_(ss, businessType, items);

    SpreadsheetApp.flush();

    return {
      success: true,
      message: `${businessType} の入力項目設定と回答シートを正常に更新・同期しました。`
    };
  });
}

/**
 * 理由マスタの保存（項目別マップ構造に対応）
 * @param {string} businessType 
 * @param {Object} reasonsData { [itemKey]: Array<Reason> }
 * @return {Object}
 */
function saveReasonMaster(businessType, reasonsData) {

      return { success: false, message: '不正なパラメータです。' };
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    saveReasons_(ss, businessType, reasonsData);

    SpreadsheetApp.flush();

    return {
      success: true,
      message: `${businessType} の未提案理由設定を正常に更新しました。`
    };
  });
}

/**
 * スタッフマスタの一括保存
 * @param {Array<Object>} staffList 
 * @return {Object}
 */
function saveStaffMaster(staffList) {
  return withLock_(() => {
    if (!Array.isArray(staffList)) {
      return { success: false, message: '不正なスタッフデータです。' };
    }

    // スタッフIDの半角英数字チェック
    for (const st of staffList) {
      const id = String(st.id || '').trim();
      if (!id) {
        return { success: false, message: 'スタッフIDが空の行があります。' };
      }
      if (!/^[A-Za-z0-9]+$/.test(id)) {
        return { success: false, message: `スタッフID「${id}」の形式が不正です（半角英数字のみ有効）。` };
      }
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    saveStaffList_(ss, staffList);

    SpreadsheetApp.flush();

    return {
      success: true,
      message: 'スタッフ情報を正常に保存しました。'
    };
  });
}

/**
 * 接客実績画面の初期データ取得
 * @param {string} staffId 
 * @return {Object}
 */
function getInitialData(staffId) {
  try {
    if (!staffId) {
      return { success: false, message: 'スタッフIDが指定されていません。' };
    }

    const cleanStaffId = String(staffId).trim();
    if (!/^[A-Za-z0-9]+$/.test(cleanStaffId)) {
      return { success: false, message: 'スタッフIDの形式が不正です（半角英数字のみ有効）。' };
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const staff = findStaffById_(ss, cleanStaffId);

    if (!staff) {
      return { success: false, message: `指定されたスタッフID（${cleanStaffId}）が見つかりません。` };
    }
    if (!staff.isActive) {
      return { success: false, message: '指定されたスタッフアカウントは現在無効化されています。管理者にお問い合わせください。' };
    }

    // 有効な項目および項目別の理由マスタを取得
    const itemsMap = loadItems_(ss, { businessType: staff.businessType, includeInactive: false });
    const fields = itemsMap[staff.businessType] || [];

    const reasonsMap = loadReasons_(ss, { businessType: staff.businessType, includeInactive: false });
    const reasonsByItem = (reasonsMap && reasonsMap[staff.businessType]) ? reasonsMap[staff.businessType] : {};

    return {
      success: true,
      data: {
        staff: staff,
        fields: fields,
        reasonsByItem: reasonsByItem,
        theme: CONFIG.THEMES[staff.businessType] || null
      }
    };

  } catch (error) {
    console.error('getInitialData エラー:', error);
    return { success: false, message: '初期データの読み込み中にエラーが発生しました: ' + error.message };
  }
}

/**
 * 接客実績データの登録処理
 * @param {Object} formData 
 * @return {Object}
 */
function submitServiceRecord(formData) {
  return withLock_(() => {
    if (!formData) {
      return { success: false, message: '不正な送信データです。' };
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // 1. スタッフ情報のサーバー再検証（改ざん防止）
    const staffId = String((formData.staff && formData.staff.id) || formData.staffId || '').trim();
    if (!staffId) {
      return { success: false, message: 'スタッフIDが指定されていません。' };
    }
    if (!/^[A-Za-z0-9]+$/.test(staffId)) {
      return { success: false, message: 'スタッフIDの形式が不正です。' };
    }

    const verifiedStaff = findStaffById_(ss, staffId);
    if (!verifiedStaff) {
      return { success: false, message: `スタッフID「${staffId}」が見つかりません。` };
    }
    if (!verifiedStaff.isActive) {
      return { success: false, message: 'スタッフアカウントが無効化されています。管理者にご確認ください。' };
    }

    const sheetName = '回答_' + verifiedStaff.businessType;
    let targetSheet = ss.getSheetByName(sheetName);
    if (!targetSheet) {
      return { success: false, message: `保存先シート「${sheetName}」が存在しません。` };
    }

    // 2. 二重送信防止（送信ID / UUID 照合）
    const submissionId = String(formData.submissionId || '').trim();
    if (!submissionId) {
      return { success: false, message: '送信ID（UUID）が存在しません。' };
    }

    const cache = CacheService.getScriptCache();
    if (cache.get('sub_' + submissionId)) {
      return { success: true, message: '接客実績を正常に登録しました。（重複防止）' };
    }

    const lastCol = targetSheet.getLastColumn();
    if (lastCol === 0) {
      return { success: false, message: '回答シートに列が存在しません。' };
    }

    const headerValues = targetSheet.getRange(1, 1, 1, lastCol).getValues()[0].map(h => String(h || '').trim());
    const submissionIdCol = headerValues.indexOf('送信ID') + 1;

    if (submissionIdCol > 0) {
      const lastRow = targetSheet.getLastRow();
      if (lastRow > 1) {
        const checkStart = Math.max(2, lastRow - 49);
        const checkCount = lastRow - checkStart + 1;
        const recentSubmissionIds = targetSheet.getRange(checkStart, submissionIdCol, checkCount, 1)
          .getValues()
          .flat()
          .map(v => String(v || '').trim());

        if (recentSubmissionIds.includes(submissionId)) {
          cache.put('sub_' + submissionId, '1', 600);
          return { success: true, message: '接客実績を正常に登録しました。（重複防止）' };
        }
      }
    }

    // 3. マスタ定義による入力値バリデーション
    const itemsMap = loadItems_(ss, { businessType: verifiedStaff.businessType, includeInactive: false });
    const fields = itemsMap[verifiedStaff.businessType] || [];
    const validItemsMap = {};

    fields.forEach(field => {
      validItemsMap[field.key] = {
        label: field.label,
        options: field.options.map(o => o.text),
        reasonOptions: new Set(field.options.filter(o => o.isReasonTrigger).map(o => o.text)),
        amountOptions: new Set(field.options.filter(o => o.isAmountTrigger).map(o => o.text))
      };
    });

    const reasonsMap = loadReasons_(ss, { businessType: verifiedStaff.businessType, includeInactive: false });
    const reasonsByItem = (reasonsMap && reasonsMap[verifiedStaff.businessType]) ? reasonsMap[verifiedStaff.businessType] : {};

    const answers = formData.answers || {};
    const reasons = formData.reasons || {};
    let requiresAmount = false;

    for (const itemKey in validItemsMap) {
      const itemInfo = validItemsMap[itemKey];
      if (itemInfo.options.length === 0) continue;

      const selectedVal = answers[itemKey];
      if (!selectedVal) {
        return { success: false, message: `「${itemInfo.label}」を選択してください。` };
      }
      if (!itemInfo.options.includes(selectedVal)) {
        return { success: false, message: `「${itemInfo.label}」の選択値「${selectedVal}」は無効です。` };
      }

      // 未提案理由バリデーション（項目別）
      if (itemInfo.reasonOptions.has(selectedVal)) {
        const itemReason = String(reasons[itemKey] || '').trim();
        const validReasonsList = (reasonsByItem[itemKey] || []).map(r => r.text);
        
        if (!itemReason) {
          return { success: false, message: `「${itemInfo.label}」の未提案理由を選択してください。` };
        }
        if (validReasonsList.length > 0 && !validReasonsList.includes(itemReason)) {
          return { success: false, message: `「${itemInfo.label}」の未提案理由「${itemReason}」はマスタに登録されていません。` };
        }
      }

      // 金額入力トリガー（買取大吉のみ）
      if (verifiedStaff.businessType === '買取大吉' && itemInfo.amountOptions.has(selectedVal)) {
        requiresAmount = true;
      }
    }

    // 金額バリデーション（0以上の整数）
    if (verifiedStaff.businessType === '買取大吉' && requiresAmount) {
      const amt = formData.amount;
      if (!isValidIntegerAmount_(amt)) {
        return { success: false, message: '買取金額を0以上の整数で入力してください。' };
      }
    }

    // 4. 回答シートへの行追加（ヘッダー文字列ベースの完全同期）
    const timestampStr = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy/MM/dd HH:mm:ss');
    const targetDateStr = formData.targetDate 
      ? String(formData.targetDate).trim().replace(/-/g, '/') 
      : Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy/MM/dd');
    const newRow = [];

    // ヘッダー名と項目キーのマッピング辞書を作成
    const labelToKeyMap = {};
    const reasonLabelToKeyMap = {};
    fields.forEach(f => {
      labelToKeyMap[f.label] = f.key;
      reasonLabelToKeyMap[`${f.label}_未提案理由`] = f.key;
    });

    for (let c = 0; c < headerValues.length; c++) {
      const colHeader = headerValues[c];

      if (colHeader === 'タイムスタンプ') {
        newRow.push(timestampStr);
      } else if (colHeader === '対応日') {
        newRow.push(targetDateStr);
      } else if (colHeader === 'スタッフID') {
        newRow.push(verifiedStaff.id);
      } else if (colHeader === 'スタッフ名') {
        newRow.push(verifiedStaff.name);
      } else if (colHeader === '店舗名') {
        newRow.push(verifiedStaff.storeName);
      } else if (colHeader === '買取金額') {
        if (verifiedStaff.businessType === '買取大吉' && requiresAmount && formData.amount !== null && formData.amount !== undefined && formData.amount !== '') {
          newRow.push(Number(formData.amount));
        } else {
          newRow.push('');
        }
      } else if (colHeader === '送信ID') {
        newRow.push(submissionId);
      } else if (labelToKeyMap[colHeader]) {
        const itemKey = labelToKeyMap[colHeader];
        newRow.push(answers[itemKey] || '');
      } else if (reasonLabelToKeyMap[colHeader]) {
        const itemKey = reasonLabelToKeyMap[colHeader];
        const selectedVal = answers[itemKey];
        const isTriggered = validItemsMap[itemKey] && validItemsMap[itemKey].reasonOptions.has(selectedVal);
        newRow.push(isTriggered ? (reasons[itemKey] || '') : '');
      } else {
        newRow.push('');
      }
    }

    targetSheet.appendRow(newRow);

    // キャッシュに submissionId を登録（600秒）
    cache.put('sub_' + submissionId, '1', 600);

    SpreadsheetApp.flush();

    return {
      success: true,
      message: '接客実績を正常に登録しました。'
    };
  });
}

    const reasonsByItem = (reasonsMap && reasonsMap[verifiedStaff.businessType]) ? reasonsMap[verifiedStaff.businessType] : {};

    for (let i = 0; i < records.length; i++) {
      const rec = records[i];
      const recAnswers = rec.answers || {};
      const recReasons = rec.reasons || {};
      let requiresAmount = false;

      for (const itemKey in validItemsMap) {
        const itemInfo = validItemsMap[itemKey];
        if (itemInfo.options.length === 0) continue;

        const selectedVal = recAnswers[itemKey];
        if (!selectedVal) {
          return { success: false, message: `#${i + 1}行目: 「${itemInfo.label}」を選択してください。` };
        }
        if (!itemInfo.options.includes(selectedVal)) {
          return { success: false, message: `#${i + 1}行目: 「${itemInfo.label}」の選択値「${selectedVal}」は無効です。` };
        }

        // 未提案理由チェック
        if (itemInfo.reasonOptions.has(selectedVal)) {
          const itemReason = String(recReasons[itemKey] || '').trim();
          const validReasonsList = (reasonsByItem[itemKey] || []).map(r => r.text);
          if (!itemReason) {
            return { success: false, message: `#${i + 1}行目: 「${itemInfo.label}」の未提案理由を選択してください。` };
          }
          if (validReasonsList.length > 0 && !validReasonsList.includes(itemReason)) {
            return { success: false, message: `#${i + 1}行目: 「${itemInfo.label}」の未提案理由「${itemReason}」はマスタに登録されていません。` };
          }
        }

        // 金額入力チェック
        if (verifiedStaff.businessType === '買取大吉' && itemInfo.amountOptions.has(selectedVal)) {
          requiresAmount = true;
        }
      }

      if (verifiedStaff.businessType === '買取大吉' && requiresAmount) {
        if (!isValidIntegerAmount_(rec.amount)) {
          return { success: false, message: `#${i + 1}行目: 買取金額を0以上の整数で入力してください。` };
        }
      }
    }

    // 4. 行データの作成と一括書き込み
    const timestampStr = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy/MM/dd HH:mm:ss');
    const labelToKeyMap = {};
    const reasonLabelToKeyMap = {};
    fields.forEach(f => {
      labelToKeyMap[f.label] = f.key;
      reasonLabelToKeyMap[`${f.label}_未提案理由`] = f.key;
    });

    const rowsToInsert = [];

    records.forEach((rec, idx) => {
      const recAnswers = rec.answers || {};
      const recReasons = rec.reasons || {};
      const row = [];

      let requiresAmount = false;
      if (verifiedStaff.businessType === '買取大吉') {
        for (const itemKey in validItemsMap) {
          const itemInfo = validItemsMap[itemKey];
          const selectedVal = recAnswers[itemKey];
          if (itemInfo.amountOptions.has(selectedVal)) {
            requiresAmount = true;
            break;
          }
        }
      }

      for (let c = 0; c < headerValues.length; c++) {
        const colHeader = headerValues[c];

        if (colHeader === 'タイムスタンプ') {
          row.push(timestampStr);
        } else if (colHeader === '対応日') {
          row.push(targetDateStr);
        } else if (colHeader === 'スタッフID') {
          row.push(verifiedStaff.id);
        } else if (colHeader === 'スタッフ名') {
          row.push(verifiedStaff.name);
        } else if (colHeader === '店舗名') {
          row.push(verifiedStaff.storeName);
        } else if (colHeader === '買取金額') {
          if (verifiedStaff.businessType === '買取大吉' && requiresAmount && rec.amount !== null && rec.amount !== undefined && rec.amount !== '') {
            row.push(Number(rec.amount));
          } else {
            row.push('');
          }
        } else if (colHeader === '送信ID') {
          row.push(`${batchId}#${idx + 1}`);
        } else if (labelToKeyMap[colHeader]) {
          const itemKey = labelToKeyMap[colHeader];
          row.push(recAnswers[itemKey] || '');
        } else if (reasonLabelToKeyMap[colHeader]) {
          const itemKey = reasonLabelToKeyMap[colHeader];
          const selectedVal = recAnswers[itemKey];
          const isTriggered = validItemsMap[itemKey] && validItemsMap[itemKey].reasonOptions.has(selectedVal);
          row.push(isTriggered ? (recReasons[itemKey] || '') : '');
        } else {
          row.push('');
        }
      }

      rowsToInsert.push(row);
    });

    const nextRow = targetSheet.getLastRow() + 1;
    targetSheet.getRange(nextRow, 1, rowsToInsert.length, headerValues.length).setValues(rowsToInsert);

    cache.put('batch_' + batchId, '1', 600);
    SpreadsheetApp.flush();

    return {
      success: true,
      count: records.length,
      message: `${records.length}件の接客実績を一括登録しました。`
    };
  });
}

/**
 * 日次集計データの登録処理（方式B：日次サマリー集計入力用）
 * @param {Object} summaryData 
 * @return {Object}
 */
function submitDailySummaryRecord(summaryData) {
  return withLock_(() => {
    if (!summaryData) {
      return { success: false, message: '不正な送信データです。' };
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // 1. スタッフ情報のサーバー再検証（改ざん防止）
    const staffId = String(summaryData.staffId || (summaryData.staff && summaryData.staff.id) || '').trim();
    if (!staffId) {
      return { success: false, message: 'スタッフIDが指定されていません。' };
    }
    if (!/^[A-Za-z0-9]+$/.test(staffId)) {
      return { success: false, message: 'スタッフIDの形式が不正です。' };
    }

    const verifiedStaff = findStaffById_(ss, staffId);
    if (!verifiedStaff) {
      return { success: false, message: `スタッフID「${staffId}」が見つかりません。` };
    }
    if (!verifiedStaff.isActive) {
      return { success: false, message: 'スタッフアカウントが無効化されています。管理者にご確認ください。' };
    }

    // 2. 二重送信防止（submissionId / UUID 照合）
    const submissionId = String(summaryData.submissionId || '').trim();
    if (!submissionId) {
      return { success: false, message: '送信ID（UUID）が存在しません。' };
    }

    const cache = CacheService.getScriptCache();
    if (cache.get('sum_' + submissionId)) {
      return { success: true, message: '日次集計を正常に登録しました。（重複防止）' };
    }

    // 保存先シートの取得
    const sheetName = verifiedStaff.businessType === '整骨院' 
      ? CONFIG.SHEETS.SUMMARY_SEIKOTSU 
      : CONFIG.SHEETS.SUMMARY_DAIKICHI;
    let targetSheet = ss.getSheetByName(sheetName);
    if (!targetSheet) {
      return { 
        success: false, 
        message: `保存先シート「${sheetName}」が見つかりません。スプレッドシートにシートが存在することをご確認ください。` 
      };
    }

    const lastCol = targetSheet.getLastColumn();
    if (lastCol === 0) {
      return { success: false, message: `集計シート「${sheetName}」に列が存在しません。` };
    }

    const headerValues = targetSheet.getRange(1, 1, 1, lastCol).getValues()[0].map(h => String(h || '').trim());
    const submissionIdCol = headerValues.indexOf('送信ID') + 1;

    if (submissionIdCol > 0) {
      const lastRow = targetSheet.getLastRow();
      if (lastRow > 1) {
        const checkStart = Math.max(2, lastRow - 49);
        const checkCount = lastRow - checkStart + 1;
        const recentSubmissionIds = targetSheet.getRange(checkStart, submissionIdCol, checkCount, 1)
          .getValues()
          .flat()
          .map(v => String(v || '').trim());

        if (recentSubmissionIds.includes(submissionId)) {
          cache.put('sum_' + submissionId, '1', 600);
          return { success: true, message: '日次集計を正常に登録しました。（重複防止）' };
        }
      }
    }

    // 3. 入力値バリデーション
    const totalCount = summaryData.totalCount;
    if (totalCount === null || totalCount === undefined || totalCount === '' || !Number.isInteger(Number(totalCount)) || Number(totalCount) < 0) {
      return { success: false, message: '本日の総対応件数を0以上の整数で入力してください。' };
    }

    const targetDateStr = summaryData.targetDate 
      ? String(summaryData.targetDate).trim().replace(/-/g, '/') 
      : Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy/MM/dd');

    const itemsMap = loadItems_(ss, { businessType: verifiedStaff.businessType, includeInactive: false });
    const fields = itemsMap[verifiedStaff.businessType] || [];
    const reasonsMap = loadReasons_(ss, { businessType: verifiedStaff.businessType, includeInactive: false });
    const reasonsByItem = (reasonsMap && reasonsMap[verifiedStaff.businessType]) ? reasonsMap[verifiedStaff.businessType] : {};

    const counts = summaryData.counts || {};
    const reasonCounts = summaryData.reasonCounts || {};

    let totalAmountTriggerCount = 0;

    // 各項目ごとに、未提案件数と未提案理由内訳合計の厳密な一致チェック
    for (const field of fields) {
      const fieldKey = field.key;
      const fCounts = counts[fieldKey] || counts[field.label] || {};
      const fReasonCounts = reasonCounts[fieldKey] || reasonCounts[field.label] || {};

      // 未提案トリガーの選択肢を集計
      let unproposedCount = 0;
      (field.options || []).forEach(opt => {
        const rawOptVal = fCounts[opt.text];
        const optCount = (rawOptVal === '' || rawOptVal === null || rawOptVal === undefined) ? 0 : Number(rawOptVal);
        if (optCount < 0 || !Number.isInteger(optCount)) {
          throw new Error(`「${field.label} - ${opt.text}」の件数は0以上の整数で入力してください。`);
        }
        if (opt.isReasonTrigger) {
          unproposedCount += optCount;
        }
        if (opt.isAmountTrigger) {
          totalAmountTriggerCount += optCount;
        }
      });

      // 未提案理由の内訳合計を集計
      let reasonSum = 0;
      const validReasons = reasonsByItem[fieldKey] || [];
      validReasons.forEach(r => {
        const rawRVal = fReasonCounts[r.text];
        const rCount = (rawRVal === '' || rawRVal === null || rawRVal === undefined) ? 0 : Number(rawRVal);
        if (rCount < 0 || !Number.isInteger(rCount)) {
          throw new Error(`「${field.label} - ${r.text}」の件数は0以上の整数で入力してください。`);
        }
        reasonSum += rCount;
      });

      // 厳密な整合性チェック
      if (unproposedCount !== reasonSum) {
        return { 
          success: false, 
          message: `「${field.label}」の未提案数（${unproposedCount}件）と理由内訳の合計（${reasonSum}件）が一致していません。` 
        };
      }
    }

    // 買取大吉の場合の金額チェック
    if (verifiedStaff.businessType === '買取大吉' && totalAmountTriggerCount > 0) {
      const totalAmt = summaryData.totalAmount;
      if (!isValidIntegerAmount_(totalAmt)) {
        return { success: false, message: '成約が1件以上あるため、買取成約総額を0以上の整数で入力してください。' };
      }
    }

    // 4. ヘッダー名に対応する値を動的マッピングして1行追加
    const timestampStr = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy/MM/dd HH:mm:ss');
    const newRow = [];

    for (let c = 0; c < headerValues.length; c++) {
      const colHeader = headerValues[c];

      if (colHeader === 'タイムスタンプ') {
        newRow.push(timestampStr);
      } else if (colHeader === '対応日') {
        newRow.push(targetDateStr);
      } else if (colHeader === 'スタッフID') {
        newRow.push(verifiedStaff.id);
      } else if (colHeader === 'スタッフ名') {
        newRow.push(verifiedStaff.name);
      } else if (colHeader === '店舗名') {
        newRow.push(verifiedStaff.storeName);
      } else if (colHeader === '本日の総対応件数') {
        newRow.push(Number(totalCount) || 0);
      } else if (colHeader === '買取成約総額') {
        if (verifiedStaff.businessType === '買取大吉' && totalAmountTriggerCount > 0 && summaryData.totalAmount !== null && summaryData.totalAmount !== undefined && summaryData.totalAmount !== '') {
          newRow.push(Number(summaryData.totalAmount));
        } else {
          newRow.push(0);
        }
      } else if (colHeader === '送信ID') {
        newRow.push(submissionId);
      } else {
        let matched = false;
        let cellVal = 0;

        // 1. 理由カラムのチェック: `${field.label}_理由_${r.text}`
        for (const field of fields) {
          const prefix = `${field.label}_理由_`;
          if (colHeader.startsWith(prefix)) {
            const rText = colHeader.slice(prefix.length).trim();
            const fReasonCounts = reasonCounts[field.key] || reasonCounts[field.label] || {};
            const rawVal = fReasonCounts[rText];
            cellVal = (rawVal === '' || rawVal === null || rawVal === undefined) ? 0 : Number(rawVal);
            matched = true;
            break;
          }
        }

        // 2. 選択肢カラムのチェック: `${field.label}_${opt.text}`
        if (!matched) {
          for (const field of fields) {
            const prefix = `${field.label}_`;
            if (colHeader.startsWith(prefix)) {
              const optText = colHeader.slice(prefix.length).trim();
              const fCounts = counts[field.key] || counts[field.label] || {};
              const rawVal = fCounts[optText];
              cellVal = (rawVal === '' || rawVal === null || rawVal === undefined) ? 0 : Number(rawVal);
              matched = true;
              break;
            }
          }
        }

        newRow.push(matched ? cellVal : '');
      }
    }

    targetSheet.appendRow(newRow);

    cache.put('sum_' + submissionId, '1', 600);
    SpreadsheetApp.flush();

    return {
      success: true,
      message: '日次集計を正常に登録しました。'
    };
  });
}

/**
 * スタッフマスタ一覧の取得
 * @param {Spreadsheet} ss 
 * @param {boolean} activeOnly 有効スタッフのみ取得するか
 * @return {Array<Object>}
 */
function loadStaffList_(ss, activeOnly = false) {
  const sheet = ss.getSheetByName(CONFIG.SHEETS.STAFF);
  if (!sheet) return [];

  const webAppUrl = getWebAppUrl_();
  const data = sheet.getDataRange().getValues();
  const staffList = [];

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const id = String(row[0] || '').trim();
    if (!id) continue;

    const isActive = toBool_(row[4], true);
    if (activeOnly && !isActive) continue;

    // 専用URL: シートの文字列または動的生成
    let url = String(row[5] || '').trim();
    if (!url && webAppUrl) {
      url = `${webAppUrl}?id=${encodeURIComponent(id)}`;
    }

    staffList.push({
      id: id,
      name: String(row[1] || '').trim(),
      businessType: String(row[2] || '').trim(),
      storeName: String(row[3] || '').trim(),
      isActive: isActive,
      url: url
    });
  }
  return staffList;
}

/**
 * 指定IDのスタッフを取得・検証
 * @param {Spreadsheet} ss 
 * @param {string} staffId 
 * @return {Object|null}
 */
function findStaffById_(ss, staffId) {
  if (!staffId) return null;
  const allStaff = loadStaffList_(ss, false);
  return allStaff.find(st => st.id === String(staffId).trim()) || null;
}

/**
 * スタッフマスタの一括保存
 * @param {Spreadsheet} ss 
 * @param {Array<Object>} staffList 
 */
function saveStaffList_(ss, staffList) {
  let sheet = ss.getSheetByName(CONFIG.SHEETS.STAFF);
  const headers = CONFIG.HEADERS.STAFF;

  if (!sheet) {
    sheet = ss.insertSheet(CONFIG.SHEETS.STAFF);
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  }

  const webAppUrl = getWebAppUrl_();
  const rows = [headers];

  staffList.forEach((st) => {
    const id = String(st.id || '').trim();
    if (!id) return;
    if (!/^[A-Za-z0-9]+$/.test(id)) {
      throw new Error(`スタッフID「${id}」は無効です。半角英数字のみ使用できます。`);
    }

    const name = String(st.name || '').trim();
    const bType = String(st.businessType || '').trim();
    const store = String(st.storeName || '').trim();
    const isActive = st.isActive !== false;

    const r = rows.length + 1;
    const formula = webAppUrl 
      ? `=IF(ISBLANK(A${r}), "", "${webAppUrl}?id=" & A${r})`
      : (st.url || '');

    rows.push([id, name, bType, store, isActive, formula]);
  });

  // A列を書式なしテキストに設定してゼロ落ちを防止
  sheet.getRange(1, 1, Math.max(sheet.getMaxRows(), rows.length), 1).setNumberFormat('@');
  overwriteSheet_(sheet, rows, headers.length);
}

/**
 * 設定_項目マスタの読み込み（8列構成対応）
 * 列順: ['事業区分', '入力項目キー', '項目表示名', '選択肢テキスト', '表示順', '未提案トリガーフラグ', '金額入力トリガーフラグ', '有効フラグ']
 * @param {Spreadsheet} ss 
 * @param {Object} options { businessType: string|null, includeInactive: boolean }
 * @return {Object} { [businessType]: Array<Item> }
 */
function loadItems_(ss, options = {}) {
  const { businessType = null, includeInactive = false } = options;
  const sheet = ss.getSheetByName(CONFIG.SHEETS.ITEMS);
  const result = {};
  CONFIG.BUSINESS_TYPES.forEach(bt => { result[bt] = []; });

  if (!sheet) return result;

  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return result;

  const headers = data[0].map(h => String(h || '').trim());
  const idxBt = headers.indexOf('事業区分') !== -1 ? headers.indexOf('事業区分') : 0;
  const idxKey = headers.indexOf('入力項目キー') !== -1 ? headers.indexOf('入力項目キー') : 1;
  const idxLabel = headers.indexOf('項目表示名') !== -1 ? headers.indexOf('項目表示名') : 2;
  const idxOptText = headers.indexOf('選択肢テキスト') !== -1 ? headers.indexOf('選択肢テキスト') : 3;
  const idxOrder = headers.indexOf('表示順') !== -1 ? headers.indexOf('表示順') : (headers.indexOf('選択肢表示順') !== -1 ? headers.indexOf('選択肢表示順') : 4);
  const idxReason = headers.indexOf('未提案トリガーフラグ') !== -1 ? headers.indexOf('未提案トリガーフラグ') : (headers.indexOf('未提案トリガー') !== -1 ? headers.indexOf('未提案トリガー') : 5);
  const idxAmount = headers.indexOf('金額入力トリガーフラグ') !== -1 ? headers.indexOf('金額入力トリガーフラグ') : (headers.indexOf('金額入力トリガー') !== -1 ? headers.indexOf('金額入力トリガー') : 6);
  const idxActive = headers.indexOf('有効フラグ') !== -1 ? headers.indexOf('有効フラグ') : 7;

  const tempMap = {};
  const orderTracker = {};
  CONFIG.BUSINESS_TYPES.forEach(bt => {
    tempMap[bt] = {};
    orderTracker[bt] = [];
  });

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const bt = String(row[idxBt] || '').trim();
    if (businessType && bt !== businessType) continue;
    if (!CONFIG.BUSINESS_TYPES.includes(bt)) continue;

    const key = String(row[idxKey] || '').trim();
    if (!bt || !key) continue;

    const optText = String(row[idxOptText] || '').trim();
    if (!optText) continue; // 選択肢テキストが空の行はスキップ

    const isActive = toBool_(row[idxActive], true);
    if (!includeInactive && !isActive) continue;

    const label = String(row[idxLabel] || '').trim();
    const optOrder = Number(row[idxOrder]) || 0;
    const isReasonTrigger = toBool_(row[idxReason], false);
    // 整骨院では金額トリガーを強制無効化
    const isAmountTrigger = (bt === '整骨院') ? false : toBool_(row[idxAmount], false);

    if (!tempMap[bt][key]) {
      orderTracker[bt].push(key);
      tempMap[bt][key] = {
        key: key,
        label: label || key,
        originalLabel: label || key,
        order: orderTracker[bt].length,
        isActive: isActive,
        options: []
      };
    } else {
      if (!isActive) tempMap[bt][key].isActive = false;
      if (label && !tempMap[bt][key].label) tempMap[bt][key].label = label;
    }

    tempMap[bt][key].options.push({
      text: optText,
      order: optOrder || (tempMap[bt][key].options.length + 1),
      isReasonTrigger: isReasonTrigger,
      isAmountTrigger: isAmountTrigger
    });
  }

  CONFIG.BUSINESS_TYPES.forEach(bt => {
    result[bt] = orderTracker[bt].map((key, idx) => {
      const item = tempMap[bt][key];
      item.order = idx + 1;
      item.options.sort((a, b) => a.order - b.order);
      return item;
    });
  });

  return result;
}

/**
 * 項目マスタの保存（8列構成で書き戻し、他事業区分のデータを維持して安全に上書き）
 * @param {Spreadsheet} ss 
 * @param {string} targetBusinessType

function saveItems_(ss, targetBusinessType, items) {

  const headers = CONFIG.HEADERS.ITEMS;

  if (!sheet) {
    sheet = ss.insertSheet(CONFIG.SHEETS.ITEMS);
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  }

  const existingData = sheet.getDataRange().getValues();
  const otherRows = [];

  if (existingData.length > 1) {
    const existingHeaders = existingData[0].map(h => String(h || '').trim());
    const idxBt = existingHeaders.indexOf('事業区分') !== -1 ? existingHeaders.indexOf('事業区分') : 0;
    const idxKey = existingHeaders.indexOf('入力項目キー') !== -1 ? existingHeaders.indexOf('入力項目キー') : 1;
    const idxLabel = existingHeaders.indexOf('項目表示名') !== -1 ? existingHeaders.indexOf('項目表示名') : 2;
    const idxOptText = existingHeaders.indexOf('選択肢テキスト') !== -1 ? existingHeaders.indexOf('選択肢テキスト') : 3;
    const idxOrder = existingHeaders.indexOf('表示順') !== -1 ? existingHeaders.indexOf('表示順') : (existingHeaders.indexOf('選択肢表示順') !== -1 ? existingHeaders.indexOf('選択肢表示順') : 4);
    const idxReason = existingHeaders.indexOf('未提案トリガーフラグ') !== -1 ? existingHeaders.indexOf('未提案トリガーフラグ') : 5;
    const idxAmount = existingHeaders.indexOf('金額入力トリガーフラグ') !== -1 ? existingHeaders.indexOf('金額入力トリガーフラグ') : 6;
    const idxActive = existingHeaders.indexOf('有効フラグ') !== -1 ? existingHeaders.indexOf('有効フラグ') : 7;

    for (let i = 1; i < existingData.length; i++) {
      const row = existingData[i];
      const bt = String(row[idxBt] || '').trim();
      if (bt && bt !== targetBusinessType) {
        otherRows.push([
          bt,
          String(row[idxKey] || '').trim(),
          String(row[idxLabel] || '').trim(),
          String(row[idxOptText] || '').trim(),
          Number(row[idxOrder]) || 1,
          toBool_(row[idxReason], false),
          bt === '整骨院' ? false : toBool_(row[idxAmount], false),
          toBool_(row[idxActive], true)
        ]);
      }
    }
  }

  const newTargetRows = [];
  items.forEach((item) => {
    const itemKey = String(item.key || '').trim();
    const itemLabel = String(item.label || itemKey).trim();
    const isActive = item.isActive !== false;
    const options = Array.isArray(item.options) && item.options.length > 0 
      ? item.options 
      : [{ text: '未設定', order: 1, isReasonTrigger: false, isAmountTrigger: false }];

    options.forEach((opt, optIdx) => {
      const optText = String(opt.text || '').trim();
      if (!optText) return;
      const optOrder = Number(opt.order) || (optIdx + 1);
      const isReason = toBool_(opt.isReasonTrigger, false);
      const isAmount = targetBusinessType === '整骨院' ? false : toBool_(opt.isAmountTrigger, false);

      newTargetRows.push([
        targetBusinessType,
        itemKey,
        itemLabel,
        optText,
        optOrder,
        isReason,
        isAmount,
        isActive
      ]);
    });
  });

  const allRows = [headers, ...otherRows, ...newTargetRows];
  overwriteSheet_(sheet, allRows, headers.length);
}

/**

function loadReasons_(ss, options = {}) {

  const sheet = ss.getSheetByName(CONFIG.SHEETS.REASONS);
  const result = {};
  CONFIG.BUSINESS_TYPES.forEach(bt => { result[bt] = {}; });

  if (!sheet) return result;

  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return result;

  const headers = data[0].map(h => String(h || '').trim());
  const idxBt = headers.indexOf('事業区分') !== -1 ? headers.indexOf('事業区分') : 0;
  const idxKey = headers.indexOf('入力項目キー') !== -1 ? headers.indexOf('入力項目キー') : 1;
  const idxReason = headers.indexOf('未提案理由') !== -1 ? headers.indexOf('未提案理由') : 2;

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const bt = String(row[idxBt] || '').trim();
    if (businessType && bt !== businessType) continue;
    if (!CONFIG.BUSINESS_TYPES.includes(bt)) continue;

    const itemKey = String(row[idxKey] || '').trim();
    const rText = String(row[idxReason] || '').trim();
    if (!bt || !itemKey || !rText) continue;

    const isActive = toBool_(row[idxActive], true);
    if (!includeInactive && !isActive) continue;

    const order = Number(row[idxOrder]) || 0;

    if (!result[bt][itemKey]) {
      result[bt][itemKey] = [];
    }

    result[bt][itemKey].push({
      text: rText,
      order: order,
      isActive: isActive
    });
  }

  CONFIG.BUSINESS_TYPES.forEach(bt => {
    if (result[bt]) {
      Object.keys(result[bt]).forEach(key => {
        result[bt][key].sort((a, b) => a.order - b.order);
      });
    }
  });

  return result;
}

/**
 * 期待される回答シートのヘッダー列配列を算出
 * @param {string} businessType 
 * @param {Array<Object>} items 
 * @return {Array<string>}
 */
function getExpectedAnswerHeaders_(businessType, items) {
  const headers = ['タイムスタンプ', '対応日', 'スタッフID', 'スタッフ名', '店舗名'];
  const activeItems = (items || []).filter(item => item.isActive !== false);

  activeItems.forEach(item => {
    const itemLabel = String(item.label || item.key).trim();
    headers.push(itemLabel);

    const hasReason = (item.options || []).some(opt => toBool_(opt.isReasonTrigger, false));
    if (hasReason) {
      headers.push(`${itemLabel}_未提案理由`);
    }

    if (businessType === '買取大吉') {
      const hasAmount = (item.options || []).some(opt => toBool_(opt.isAmountTrigger, false));
      if (hasAmount) {
        headers.push('買取金額');
      }
    }
  });

  headers.push('送信ID');
  return headers;
}

/**
 * 回答シートのヘッダー動的同期（DeveloperMetadataを使わず安全に同期）
 * @param {Spreadsheet} ss 
 * @param {string} businessType 
 * @param {Array<Object>} items 
 */
function syncAnswerSheetHeaders_(ss, businessType, items) {
  const sheetName = '回答_' + businessType;
  let answerSheet = ss.getSheetByName(sheetName);
  if (!answerSheet) {
    answerSheet = ss.insertSheet(sheetName);
  }

  const expectedHeaders = getExpectedAnswerHeaders_(businessType, items);
  const theme = CONFIG.THEMES[businessType] || {};
  const headerBg = theme.answerHeaderBg || '#2563EB';

  const lastRow = answerSheet.getLastRow();
  const lastCol = answerSheet.getLastColumn();

  if (lastCol === 0 || lastRow <= 1) {
    // 既存回答データがない場合は完全新構成でヘッダーを設定
    if (lastCol > expectedHeaders.length) {
      answerSheet.getRange(1, expectedHeaders.length + 1, 1, lastCol - expectedHeaders.length).clearContent();
    }
    answerSheet.getRange(1, 1, 1, expectedHeaders.length).setValues([expectedHeaders]);

    const range = answerSheet.getRange(1, 1, 1, expectedHeaders.length);
    range.setBackground(headerBg);
    range.setFontColor('#FFFFFF');
    range.setFontWeight('bold');
    range.setHorizontalAlignment('center');
    range.setVerticalAlignment('middle');
    answerSheet.setRowHeight(1, 35);
    answerSheet.setFrozenRows(1);
    try {
      answerSheet.autoResizeColumns(1, expectedHeaders.length);
    } catch (e) {}
    return;
  }

  // 既存回答データがある場合：
  const existingHeaders = answerSheet.getRange(1, 1, 1, lastCol).getValues()[0].map(h => String(h || '').trim());

  // 0. 「対応日」列が未存在の場合、タイムスタンプの直後（第2列）に列挿入して安全に同期
  if (!existingHeaders.includes('対応日')) {
    const tsIdx = existingHeaders.indexOf('タイムスタンプ');
    const insertAfterCol = tsIdx !== -1 ? (tsIdx + 1) : 1;
    answerSheet.insertColumnAfter(insertAfterCol);
    const dateCell = answerSheet.getRange(1, insertAfterCol + 1);
    dateCell.setValue('対応日');
    dateCell.setBackground(headerBg);
    dateCell.setFontColor('#FFFFFF');
    dateCell.setFontWeight('bold');
    dateCell.setHorizontalAlignment('center');
    dateCell.setVerticalAlignment('middle');
    existingHeaders.splice(insertAfterCol, 0, '対応日');
  }

  // 1. 既存項目名の変更（originalLabel -> label）を反映
  items.forEach(item => {
    if (item.originalLabel && item.originalLabel !== item.label) {
      const oldIdx = existingHeaders.indexOf(item.originalLabel);
      if (oldIdx !== -1) {
        answerSheet.getRange(1, oldIdx + 1).setValue(item.label);
        existingHeaders[oldIdx] = item.label;

        const oldReasonIdx = existingHeaders.indexOf(`${item.originalLabel}_未提案理由`);
        if (oldReasonIdx !== -1) {
          answerSheet.getRange(1, oldReasonIdx + 1).setValue(`${item.label}_未提案理由`);
          existingHeaders[oldReasonIdx] = `${item.label}_未提案理由`;
        }
      }
    }
  });

  // 2. 期待される列が存在しなければ末尾に追加
  expectedHeaders.forEach(expHeader => {
    if (!existingHeaders.includes(expHeader)) {
      const newCol = answerSheet.getLastColumn() + 1;
      const cell = answerSheet.getRange(1, newCol);
      cell.setValue(expHeader);
      cell.setBackground(headerBg);
      cell.setFontColor('#FFFFFF');
      cell.setFontWeight('bold');
      cell.setHorizontalAlignment('center');
      cell.setVerticalAlignment('middle');
      existingHeaders.push(expHeader);
    }
  });

  try {
    answerSheet.autoResizeColumns(1, answerSheet.getLastColumn());
  } catch (e) {}
}

 * google.script.run の Promise 化ラッパー
 */
function callServer(funcName, ...args) {
  return new Promise((resolve, reject) => {
    if (!google || !google.script || !google.script.run) {
      reject(new Error('google.script.run が利用できません。'));
      return;
    }
    google.script.run
      .withSuccessHandler(resolve)
      .withFailureHandler(reject)[funcName](...args);
  });
}

/**
 * google.script.run の Promise 化ラッパー
 */
function callServer(funcName, ...args) {
  return new Promise((resolve, reject) => {
    if (!google || !google.script || !google.script.run) {
      reject(new Error('google.script.run が利用できません。'));
      return;
    }
    google.script.run
      .withSuccessHandler(resolve)
      .withFailureHandler(reject)[funcName](...args);
  });
}

/**
 * UUID 生成ヘルパー
 */
function generateUUID() {
  if (typeof crypto !== 'undefined' && crypto.randomUUID) {
    return crypto.randomUUID();
  }
  return 'sub_' + Date.now() + '_' + Math.random().toString(36).substring(2, 11);
}

      if (response && response.success) {
        modalTitle.value = '日次集計の登録が完了しました！';
        modalMessage.value = '本日の接客日次集計を正常に記録しました。<br>お疲れ様でした！';

        summarySubmissionId.value = generateUUID();

        setTimeout(() => {
          isCompleted.value = false;
          resetSummaryForm();
          window.scrollTo({ top: 0, behavior: 'smooth' });
        }, 1500);
      } else {
        summaryServerErrorMessage.value = response ? response.message : '日次集計の登録に失敗しました。';
      }
    } catch (error) {
      isSummarySubmitting.value = false;
      console.error('Summary Submit Error:', error);
      summaryServerErrorMessage.value = '通信エラーが発生しました: ' + (error.message || '時間をおいて再送信してください。');
    }
  };

  const adjustSummaryCount = (fieldKey, optText, delta) => {
    if (!summaryCounts[fieldKey]) summaryCounts[fieldKey] = {};
    const current = Number(summaryCounts[fieldKey][optText]) || 0;
    const next = Math.max(0, current + delta);
    summaryCounts[fieldKey][optText] = next === 0 ? '' : next;
  };

  const adjustSummaryReasonCount = (fieldKey, rText, delta) => {
    if (!summaryReasonCounts[fieldKey]) summaryReasonCounts[fieldKey] = {};
    const current = Number(summaryReasonCounts[fieldKey][rText]) || 0;
    const next = Math.max(0, current + delta);
    summaryReasonCounts[fieldKey][rText] = next === 0 ? '' : next;
  };

  return {

    currentTab,

/**
 * 真偽値の安全なパース
 * @param {*} val 判定対象
 * @param {boolean} defaultValue デフォルト値
 * @return {boolean}
 */
function toBool_(val, defaultValue = false) {
  if (val === undefined || val === null || val === '') return defaultValue;
  if (typeof val === 'boolean') return val;
  const s = String(val).trim().toUpperCase();
  if (s === 'TRUE' || s === '1') return true;
  if (s === 'FALSE' || s === '0') return false;
  return defaultValue;
}

/**
 * Webアプリ公開URLの取得
 * スクリプトプロパティ WEB_APP_URL を優先、なければ ScriptApp から取得
 * @return {string}
 */
function getWebAppUrl_() {
  try {
    const propUrl = PropertiesService.getScriptProperties().getProperty(CONFIG.PROP_WEB_APP_URL);
    if (propUrl && propUrl.trim()) {
      return propUrl.trim();
    }
    return ScriptApp.getService().getUrl() || '';
  } catch (e) {
    return '';
  }
}

/**
 * ロック取得・解放の定型ラッパー関数
 * @param {Function} fn 実行する処理
 * @param {number} timeoutMs ロック待機ミリ秒（デフォルト10秒）
 * @param {string} busyMessage 競合時のエラーメッセージ
 * @return {Object}
 */
function withLock_(fn, timeoutMs = 10000, busyMessage = '他ユーザーの書き込み処理と競合しました。時間をおいて再実行してください。') {
  const lock = LockService.getScriptLock();
  try {
    const hasLock = lock.tryLock(timeoutMs);
    if (!hasLock) {
      return { success: false, message: busyMessage };
    }
    return fn();
  } catch (error) {
    console.error('withLock_ 実行エラー:', error);
    return { success: false, message: error.message || '処理中にエラーが発生しました。' };
  } finally {
    lock.releaseLock();
  }
}

/**
 * シートの安全な上書きヘルパー
 * 書き込んだあと余剰行のみをクリアして全件消失を防止
 * @param {Sheet} sheet 対象シート
 * @param {Array<Array<*>>} rows 書き込む行データ配列
 * @param {number} width 列数
 */
function overwriteSheet_(sheet, rows, width) {
  if (!rows || rows.length === 0) return;
  const normalized = rows.map(r => Array.from({ length: width }, (_, i) => r[i] ?? ''));
  sheet.getRange(1, 1, normalized.length, width).setValues(normalized);
  const extra = sheet.getMaxRows() - normalized.length;
  if (extra > 0) {
    sheet.getRange(normalized.length + 1, 1, extra, sheet.getMaxColumns()).clearContent();
  }
}

/**
 * HTMLエスケープ処理（XSS防止）
 * @param {*} str エスケープ対象文字列
 * @return {string}
 */
function escapeHtml_(str) {
  if (str === null || str === undefined) return '';
  return String(str)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}

/**
 * 0以上の正の整数チェック
 * @param {*} val チェック対象
 * @return {boolean}
 */
function isValidIntegerAmount_(val) {
  if (val === null || val === undefined || val === '') return false;
  const num = Number(val);
  return Number.isInteger(num) && num >= 0;
}
