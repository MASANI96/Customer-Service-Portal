/**
 * 日付オブジェクトまたは文字列を標準フォーマット(yyyy/MM/dd HH:mm:ss)に変換
 */
function formatDateTime_(val) {
  if (!val) return '';
  if (val instanceof Date) {
    if (isNaN(val.getTime())) return '';
    return Utilities.formatDate(val, 'Asia/Tokyo', 'yyyy/MM/dd HH:mm:ss');
  }
  return String(val).trim();
}

/**
 * 日付オブジェクトまたは文字列から YYYY-MM-DD 形式の文字列を抽出・正規化するヘルパー
 */
function formatDateKey_(val) {
  if (!val) return '';
  if (val instanceof Date) {
    if (isNaN(val.getTime())) return '';
    return Utilities.formatDate(val, 'Asia/Tokyo', 'yyyy-MM-dd');
  }
  const str = String(val).trim();
  const match = str.match(/^(\d{4})[-/](\d{1,2})[-/](\d{1,2})/);
  if (match) {
    const y = match[1];
    const m = String(match[2]).padStart(2, '0');
    const d = String(match[3]).padStart(2, '0');
    return `${y}-${m}-${d}`;
  }
  return str;
}

/**
 * 日付オブジェクトまたは文字列から YYYY-MM 形式の文字列を抽出・正規化するヘルパー
 */
function formatYearMonth_(val) {
  if (!val) return '';
  if (val instanceof Date) {
    if (isNaN(val.getTime())) return '';
    return Utilities.formatDate(val, 'Asia/Tokyo', 'yyyy-MM');
  }
  const str = String(val).trim();
  const match = str.match(/^(\d{4})[-/](\d{1,2})/);
  if (match) {
    const y = match[1];
    const m = String(match[2]).padStart(2, '0');
    return `${y}-${m}`;
  }
  return str;
}

/**
 * 真偽値判定ヘルパー
 */
function parseBoolean_(val) {
  if (val === true || val === 1) return true;
  if (!val) return false;
  const s = String(val).trim().toLowerCase();
  return s === 'true' || s === '1' || s === 'yes' || s === 'on';
}

/**
 * シートのデータをオブジェクト配列として取得する汎用関数
 */
function getSheetDataAsObjects_(sheet) {
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  if (!data || data.length < 2) return [];

  const headers = data[0].map(h => String(h || '').trim());
  const rows = [];

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    // 空行スキップ
    const hasData = row.some(cell => cell !== '' && cell !== null && cell !== undefined);
    if (!hasData) continue;

    const rowObj = {};
    for (let j = 0; j < headers.length; j++) {
      const header = headers[j];
      if (!header) continue;
      let val = row[j];
      if (val instanceof Date) {
        val = formatDateTime_(val);
      } else if (typeof val === 'string') {
        val = val.trim();
      }
      rowObj[header] = val;
    }
    rows.push(rowObj);
  }
  return rows;
}

 * クライアントのインメモリキャッシュに利用

function getDashboardData() {
  try {
    const ss = getSpreadsheet_();

    // 1. スタッフマスタ

      item['スタッフ名'] = String(item['スタッフ名'] || '').trim();
      item['店舗名'] = String(item['店舗名'] || '').trim();

      // 有効フラグをboolean化
      item.isActive = parseBoolean_(item['有効フラグ']);
      // 専用URLはフロントでは不要
      delete item['専用URL'];
      return item;
    });

    // 2. 設定_項目マスタ
    const itemSheet = ss.getSheetByName('設定_項目マスタ');
    const itemMaster = getSheetDataAsObjects_(itemSheet).map(item => {
      item['事業区分'] = String(item['事業区分'] || '').trim();
      // シートの実際の列名（項目表示名・選択肢テキスト）を取得し、項目名・選択肢名としても保持
      const itemName = String(item['項目表示名'] || item['項目名'] || '').trim();
      const choiceText = String(item['選択肢テキスト'] || item['選択肢名'] || '').trim();
      item['項目表示名'] = itemName;
      item['項目名'] = itemName;
      item['選択肢テキスト'] = choiceText;
      item['選択肢名'] = choiceText;
      item.isActive = parseBoolean_(item['有効フラグ']);
      item.isSuccess = parseBoolean_(item['成功フラグ']);
      item.isUnproposedTrigger = parseBoolean_(item['未提案トリガーフラグ']);
      item.isPriceTrigger = parseBoolean_(item['金額入力トリガーフラグ']);
      return item;
    });

    // 3. 回答_整骨院
    const seikotsuinSheet = ss.getSheetByName('回答_整骨院');
    const seikotsuinRecords = getSheetDataAsObjects_(seikotsuinSheet).map(item => {
      // 送信IDはダッシュボード上では除外
      delete item['送信ID'];
      item['スタッフID'] = String(item['スタッフID'] || '').trim();
      item['スタッフ名'] = String(item['スタッフ名'] || '').trim();
      item['店舗名'] = String(item['店舗名'] || '').trim();
      // 対応日の標準フォーマット保持
      if (item['対応日']) {
        item['対応日'] = formatDateKey_(item['対応日']);
      }
      return item;
    });

    // 4. 回答_買取大吉
    const daikichiSheet = ss.getSheetByName('回答_買取大吉');
    const daikichiRecords = getSheetDataAsObjects_(daikichiSheet).map(item => {
      // 送信ID除外
      delete item['送信ID'];
      item['スタッフID'] = String(item['スタッフID'] || '').trim();
      item['スタッフ名'] = String(item['スタッフ名'] || '').trim();
      item['店舗名'] = String(item['店舗名'] || '').trim();
      // 対応日の標準フォーマット保持
      if (item['対応日']) {
        item['対応日'] = formatDateKey_(item['対応日']);
      }
      // 買取金額の数値パース
      let price = item['買取金額'];
      if (typeof price === 'string') {
        price = parseFloat(price.replace(/,/g, '')) || 0;
      } else if (typeof price !== 'number') {
        price = 0;
      }
      item.parsedPrice = price;
      return item;
    });

    // 5. 目標管理
    const targetSheet = ss.getSheetByName('目標管理');
    const rawTargetRecords = getSheetDataAsObjects_(targetSheet);
    
    // 年月を正規化し、同一キーの重複がある場合は最新更新日時のレコードを優先採用
    const targetMap = {};
    rawTargetRecords.forEach(item => {
      let val = item['目標値'];
      if (typeof val === 'string') {
        val = parseFloat(val.replace(/,/g, ''));
      }
      item.targetValue = (typeof val === 'number' && !isNaN(val)) ? val : 0;
      item['対象レベル'] = String(item['対象レベル'] || '全社').trim();
      item['対象名'] = String(item['対象名'] || '全社').trim();
      item['KPI項目'] = String(item['KPI項目'] || '').trim();
      item['事業区分'] = String(item['事業区分'] || '').trim();
      item['年月'] = formatYearMonth_(item['年月']);

      const uniqueKey = `${item['年月']}__${item['事業区分']}__${item['対象レベル']}__${item['対象名']}__${item['KPI項目']}`;
      
      if (!targetMap[uniqueKey]) {
        targetMap[uniqueKey] = item;
      } else {
        const existingTime = new Date(targetMap[uniqueKey]['更新日時'] || 0).getTime();
        const currentTime = new Date(item['更新日時'] || 0).getTime();
        if (currentTime >= existingTime) {
          targetMap[uniqueKey] = item;
        }
      }
    });

    const targetRecords = Object.values(targetMap);

    const nowStr = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy/MM/dd HH:mm:ss');

    return {
      status: 'success',
      serverTime: nowStr,
      data: {
        staffList: staffList,
        itemMaster: itemMaster,
        seikotsuinRecords: seikotsuinRecords,
        daikichiRecords: daikichiRecords,
        targetRecords: targetRecords
      }
    };
  } catch (err) {
    Logger.log('getDashboardData エラー: ' + err.stack);
    return {
      status: 'error',
      message: err.message || err.toString()
    };
  }
}

/**
 * 目標管理シートへの目標値保存・更新・削除API
 * LockServiceによる排他制御を導入
 * @param {Object} payload { yearMonth, businessCategory, targetLevel, targetName, targets: { [kpi]: value } }
 */
function saveKpiTargets(payload) {
  const lock = LockService.getScriptLock();
  try {
    // 30秒間ロック取得待機
    lock.waitLock(30000);

    if (!payload || !payload.yearMonth || !payload.businessCategory) {
      throw new Error('保存パラメータが不正です。');
    }

    const yearMonth = formatYearMonth_(payload.yearMonth); // 確実に YYYY-MM に正規化
    const businessCategory = String(payload.businessCategory).trim();
    const targetLevel = String(payload.targetLevel || '全社').trim();
    const targetName = String(payload.targetName || (targetLevel === '全社' ? '全社' : '')).trim();

    if ((targetLevel === '店舗' || targetLevel === 'スタッフ') && !targetName) {
      throw new Error(`対象レベルが「${targetLevel}」のとき、対象名は必須です。`);
    }

    const ss = getSpreadsheet_();
    let targetSheet = ss.getSheetByName('目標管理');
    if (!targetSheet) {
      targetSheet = ss.insertSheet('目標管理');
      targetSheet.appendRow(['年月', '事業区分', '対象レベル', '対象名', 'KPI項目', '目標値', '更新日時']);
    }

    const data = targetSheet.getDataRange().getValues();
    const headers = data[0].map(h => String(h || '').trim());

    // ヘッダーインデックス特定
    const ymIdx = headers.indexOf('年月');
    const catIdx = headers.indexOf('事業区分');
    const lvlIdx = headers.indexOf('対象レベル');
    const nameIdx = headers.indexOf('対象名');
    const kpiIdx = headers.indexOf('KPI項目');
    const valIdx = headers.indexOf('目標値');
    const dateIdx = headers.indexOf('更新日時');

    if (ymIdx === -1 || catIdx === -1 || kpiIdx === -1 || valIdx === -1) {
      throw new Error('目標管理シートのヘッダー構成が正しくありません。「年月, 事業区分, 対象レベル, 対象名, KPI項目, 目標値, 更新日時」が必要です。');
    }

    const targets = payload.targets || {};
    const nowStr = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy/MM/dd HH:mm:ss');

    // 既存行のマッピング (同一キーに複数行ある場合は配列で保持)
    const existingRowMap = {};
    for (let r = 1; r < data.length; r++) {
      const rowYm = formatYearMonth_(data[r][ymIdx]);
      const rowCat = String(data[r][catIdx] || '').trim();
      const rowLvl = lvlIdx !== -1 ? String(data[r][lvlIdx] || '').trim() : '';
      const rowName = nameIdx !== -1 ? String(data[r][nameIdx] || '').trim() : '';
      const rowKpi = String(data[r][kpiIdx] || '').trim();
      const key = `${rowYm}__${rowCat}__${rowLvl}__${rowName}__${rowKpi}`;
      if (!existingRowMap[key]) {
        existingRowMap[key] = [];
      }
      existingRowMap[key].push(r + 1); // 1-indexed 行番号
    }

    const rowsToDelete = [];
    const kpiKeys = Object.keys(targets);

    for (let k = 0; k < kpiKeys.length; k++) {
      const kpiName = String(kpiKeys[k]).trim();
      const rawVal = targets[kpiName];
      const lookupKey = `${yearMonth}__${businessCategory}__${targetLevel}__${targetName}__${kpiName}`;

      // 空欄・null・undefined の場合は「削除（取り消し）」
      if (rawVal === null || rawVal === undefined || rawVal === '') {
        if (existingRowMap[lookupKey] && existingRowMap[lookupKey].length > 0) {
          existingRowMap[lookupKey].forEach(rowNum => {
            rowsToDelete.push(rowNum);
          });
        }
        continue;
      }

      // 有効な数値（0以上）
      const targetVal = parseFloat(rawVal);
      const safeVal = isNaN(targetVal) ? 0 : targetVal;

      if (existingRowMap[lookupKey] && existingRowMap[lookupKey].length > 0) {
        // 最初の行を最新値で更新
        const primaryRowNum = existingRowMap[lookupKey][0];
        targetSheet.getRange(primaryRowNum, ymIdx + 1).setNumberFormat('@').setValue(yearMonth);
        targetSheet.getRange(primaryRowNum, valIdx + 1).setValue(safeVal);
        if (dateIdx !== -1) {
          targetSheet.getRange(primaryRowNum, dateIdx + 1).setValue(nowStr);
        }
        // 2行目以降の重複行は削除リストに追加
        for (let i = 1; i < existingRowMap[lookupKey].length; i++) {
          rowsToDelete.push(existingRowMap[lookupKey][i]);
        }
      } else {
        // 新規追加 (先頭シングルクォーテーションで日付型への自動変換を防止)
        const newRow = new Array(headers.length).fill('');
        newRow[ymIdx] = "'" + yearMonth;
        newRow[catIdx] = businessCategory;
        if (lvlIdx !== -1) newRow[lvlIdx] = targetLevel;
        if (nameIdx !== -1) newRow[nameIdx] = targetName;
        newRow[kpiIdx] = kpiName;
        newRow[valIdx] = safeVal;
        if (dateIdx !== -1) newRow[dateIdx] = nowStr;
        targetSheet.appendRow(newRow);
      }
    }

    // 削除行を下の行から降順に削除（行インデックスずれ防止）
    if (rowsToDelete.length > 0) {
      const uniqueRowsToDelete = Array.from(new Set(rowsToDelete)).sort((a, b) => b - a);
      uniqueRowsToDelete.forEach(rowNum => {
        targetSheet.deleteRow(rowNum);
      });
    }

    // 更新後の目標管理シートのデータを再取得して返却（重複排除済み）
    const updatedRaw = getSheetDataAsObjects_(targetSheet);
    const updatedMap = {};
    updatedRaw.forEach(item => {
      let val = item['目標値'];
      if (typeof val === 'string') {
        val = parseFloat(val.replace(/,/g, ''));
      }
      item.targetValue = (typeof val === 'number' && !isNaN(val)) ? val : 0;
      item['対象レベル'] = String(item['対象レベル'] || '全社').trim();
      item['対象名'] = String(item['対象名'] || '全社').trim();
      item['KPI項目'] = String(item['KPI項目'] || '').trim();
      item['事業区分'] = String(item['事業区分'] || '').trim();
      item['年月'] = formatYearMonth_(item['年月']);

      const key = `${item['年月']}__${item['事業区分']}__${item['対象レベル']}__${item['対象名']}__${item['KPI項目']}`;
      if (!updatedMap[key]) {
        updatedMap[key] = item;
      } else {
        const existingTime = new Date(updatedMap[key]['更新日時'] || 0).getTime();
        const currentTime = new Date(item['更新日時'] || 0).getTime();
        if (currentTime >= existingTime) {
          updatedMap[key] = item;
        }
      }
    });

    return {
      status: 'success',
      message: '目標値を正常に更新・保存しました。',
      updatedTargetRecords: Object.values(updatedMap)
    };
  } catch (err) {
    Logger.log('saveKpiTargets エラー: ' + err.stack);
    return {
      status: 'error',
      message: err.message || err.toString()
    };
  } finally {
    try {
      lock.releaseLock();
    } catch (e) {
      Logger.log('lock.releaseLock エラー: ' + e);
    }
  }
}

  const { createApp, ref, reactive, computed, watch, onMounted, nextTick } = Vue;

  createApp({
    setup() {
      // ----------------------------------------------------
      // 1. リアクティブステート定義
      // ----------------------------------------------------
      const isLoading = ref(true);
      const hasInitialLoaded = ref(false);
      const errorMessage = ref('');
      const lastSyncTime = ref('');

      // 現在の事業区分 ('整骨院' | '買取大吉')
      const currentCategory = ref('整骨院');

      // アクティブビュータブ ('overview' | 'store' | 'staff' | 'matrix')
      const activeViewTab = ref('overview');

      // 前期間比較モード ('previous': 直前同期間 | 'prevYear': 前年同期間)
      const comparisonMode = ref('previous');

      // 全データキャッシュ (インメモリ)
      const allData = reactive({
        staffList: [],
        itemMaster: [],
        seikotsuinRecords: [],
        daikichiRecords: [],
        targetRecords: []
      });

      // 期間選択ステート
      const presetType = ref('month'); // 'year' | 'half' | 'quarter' | 'month' | 'relative' | 'around' | 'custom'
      const presetValue = ref(0);       // 0: 今月
      const customDateFrom = ref('');
      const customDateTo = ref('');
      const activeYearForQuarters = ref(new Date().getFullYear());

      // フィルターステート
      const selectedStore = ref('all');
      const selectedStaff = ref('all');

      // 未提出者チェックステート
      const submissionCheckDate = ref(formatDateKey(new Date()));

      // チャート表示設定
      const trendGrouping = ref('daily'); // 'daily' | 'monthly'
      const trendSelectedKpi = ref('nextReservation');
      const selectedReasonItem = ref('次回予約');

      // テーブルソートステート
      const storeSortKey = ref('');
      const storeSortOrder = ref('desc');
      const staffSortKey = ref('');
      const staffSortOrder = ref('desc');

      // マトリクスビュー設定
      const matrixDimension = ref('staff'); // 'staff' | 'store'
      const matrixGranularity = ref('daily'); // 'daily' | 'weekly'
      const matrixSelectedKpi = ref('次回予約_獲得率');

      // スタッフ詳細ドロワー
      const showStaffDrawer = ref(false);
      const drawerStaffData = ref(null);

      // 目標設定モーダル
      const showTargetModal = ref(false);
      const isSavingTarget = ref(false);
      const targetModalMessage = ref('');
      const targetModalIsError = ref(false);
      const targetStaffStoreFilter = ref('all');
      const targetForm = reactive({
        yearMonth: formatYearMonth(new Date()),
        businessCategory: '整骨院',
        targetLevel: '全社', // '全社' | '店舗' | 'スタッフ'
        targetName: '全社',
        values: {}
      });

      // Chart.js インスタンス管理
      let trendChartInstance = null;
      let storeChartInstance = null;
      let reasonChartInstance = null;
      let staffTrendChartInstance = null;

      // ----------------------------------------------------
      // 2. 日付ユーティリティ関数（ローカル日付計算・UTCズレ防止）
      // ----------------------------------------------------
      function parseDateParts(dateStr) {
        if (!dateStr) return null;
        const match = String(dateStr).trim().match(/^(\d{4})[-/](\d{1,2})[-/](\d{1,2})/);
        if (!match) return null;
        return {
          y: parseInt(match[1], 10),
          m: parseInt(match[2], 10),
          d: parseInt(match[3], 10)
        };
      }

      function createLocalDate(dateStr) {
        const parts = parseDateParts(dateStr);
        if (!parts) return null;
        return new Date(parts.y, parts.m - 1, parts.d);
      }

      function formatDateKey(d) {
        if (!d) return '';
        const y = d.getFullYear();
        const m = String(d.getMonth() + 1).padStart(2, '0');
        const day = String(d.getDate()).padStart(2, '0');
        return `${y}-${m}-${day}`;
      }

      function formatYearMonth(d) {
        if (!d) return '';
        const y = d.getFullYear();
        const m = String(d.getMonth() + 1).padStart(2, '0');
        return `${y}-${m}`;
      }

      function normalizeYearMonth(val) {
        if (!val) return '';
        if (val instanceof Date) return formatYearMonth(val);
        const str = String(val).trim();
        const match = str.match(/^(\d{4})[-/](\d{1,2})/);
        if (match) {
          const y = match[1];
          const m = String(match[2]).padStart(2, '0');
          return `${y}-${m}`;
        }
        return str;
      }

      // 回答レコードから対応日（実態日）を優先抽出する
      function extractDateStr(val) {
        if (!val) return '';
        if (typeof val === 'object' && !(val instanceof Date)) {
          // レコードオブジェクトが渡された場合、対応日を最優先
          const recDate = val['対応日'] || val['タイムスタンプ'];
          return extractDateStr(recDate);
        }
        if (val instanceof Date) return formatDateKey(val);
        const str = String(val).trim();
        const match = str.match(/^(\d{4})[-/](\d{1,2})[-/](\d{1,2})/);
        if (match) {
          const y = match[1];
          const m = String(match[2]).padStart(2, '0');
          const d = String(match[3]).padStart(2, '0');
          return `${y}-${m}-${d}`;
        }
        return '';
      }

      function addDays(d, days) {
        const result = new Date(d);
        result.setDate(result.getDate() + days);
        return result;
      }

      function addYears(d, years) {
        const result = new Date(d);
        result.setFullYear(result.getFullYear() + years);
        return result;
      }

      // 2つの日付文字列 (YYYY-MM-DD) 間の実日数を計算
      function daysBetween(startStr, endStr) {
        const d1 = createLocalDate(startStr);
        const d2 = createLocalDate(endStr);
        if (!d1 || !d2) return 1;
        const diff = Math.round((d2.getTime() - d1.getTime()) / (1000 * 60 * 60 * 24)) + 1;
        return diff > 0 ? diff : 1;
      }

      // ----------------------------------------------------
      // 3. 期間計算 & 今月経過日数 & 比較期間ロジック
      // ----------------------------------------------------
      const availableYears = computed(() => {
        const currentYear = new Date().getFullYear();
        return [currentYear, currentYear - 1, currentYear - 2];
      });

      // 選択中の日付範囲 [start, end] (YYYY-MM-DD)
      const activeDateRange = computed(() => {
        const now = new Date();

        if (presetType.value === 'custom') {
          return {
            start: customDateFrom.value || '2020-01-01',
            end: customDateTo.value || '2099-12-31'
          };
        }

        if (presetType.value === 'year') {
          const y = presetValue.value;
          return { start: `${y}-01-01`, end: `${y}-12-31` };
        }

        if (presetType.value === 'half') {
          const y = activeYearForQuarters.value;
          if (presetValue.value === 1) {
            return { start: `${y}-01-01`, end: `${y}-06-30` };
          } else {
            return { start: `${y}-07-01`, end: `${y}-12-31` };
          }
        }

        if (presetType.value === 'quarter') {
          const y = activeYearForQuarters.value;
          const q = presetValue.value;
          if (q === 1) return { start: `${y}-01-01`, end: `${y}-03-31` };
          if (q === 2) return { start: `${y}-04-01`, end: `${y}-06-30` };
          if (q === 3) return { start: `${y}-07-01`, end: `${y}-09-30` };
          if (q === 4) return { start: `${y}-10-01`, end: `${y}-12-31` };
        }

        if (presetType.value === 'month') {
          const offset = presetValue.value;
          if (offset === 0) {
            // 今月 (進行中の月): 当月1日 〜 本日 (当日含む)
            const y = now.getFullYear();
            const m = now.getMonth();
            const firstDay = new Date(y, m, 1);
            return {
              start: formatDateKey(firstDay),
              end: formatDateKey(now)
            };
          } else {
            // 過去の月: 月初 〜 月末
            const target = new Date(now.getFullYear(), now.getMonth() + offset, 1);
            const y = target.getFullYear();
            const m = target.getMonth();
            const firstDay = new Date(y, m, 1);
            const lastDay = new Date(y, m + 1, 0);
            return {
              start: formatDateKey(firstDay),
              end: formatDateKey(lastDay)
            };
          }
        }

        if (presetType.value === 'relative') {
          const months = presetValue.value;
          const past = new Date(now.getFullYear(), now.getMonth() - months, now.getDate());
          return { start: formatDateKey(past), end: formatDateKey(now) };
        }

        if (presetType.value === 'around') {
          const months = presetValue.value;
          const past = new Date(now.getFullYear(), now.getMonth() - months, 1);
          const future = new Date(now.getFullYear(), now.getMonth() + months + 1, 0);
          return { start: formatDateKey(past), end: formatDateKey(future) };
        }

        return { start: '2020-01-01', end: '2099-12-31' };
      });

      const formattedDateRangeLabel = computed(() => {
        const { start, end } = activeDateRange.value;
        return `${start.replace(/-/g, '/')} 〜 ${end.replace(/-/g, '/')}`;
      });

      // 選択期間の実日数（今月の場合は経過日数）
      const dateRangeDaysCount = computed(() => {
        const { start, end } = activeDateRange.value;
        return daysBetween(start, end);
      });

      // 日平均の除算分母（今月は経過日数、それ以外は選択期間日数）
      const activeCalculationDaysCount = computed(() => {
        return dateRangeDaysCount.value;
      });

      // 月単位の選択かどうか（金額KPIの達成率計算判定用: 月以外は「-」）
      const isMonthRange = computed(() => {
        return presetType.value === 'month';
      });

      // 45日超過時は週次に固定
      watch(dateRangeDaysCount, (newDays) => {
        if (newDays > 45 && matrixGranularity.value === 'daily') {
          matrixGranularity.value = 'weekly';
        }
      });

      // 比較期間 (直前同期間 & 前年同期間)
      const previousPeriodRange = computed(() => {
        const now = new Date();
        if (presetType.value === 'month' && presetValue.value === 0) {
          // 今月の場合: 前月1日 〜 前月同日 (例: 本日が10/28なら前月は9/1〜9/28)
          const y = now.getFullYear();
          const m = now.getMonth();
          const prevMonthFirst = new Date(y, m - 1, 1);
          const prevMonthLastDay = new Date(y, m, 0).getDate();
          const targetDay = Math.min(now.getDate(), prevMonthLastDay);
          const prevMonthSameDay = new Date(y, m - 1, targetDay);
          return {
            start: formatDateKey(prevMonthFirst),
            end: formatDateKey(prevMonthSameDay)
          };
        } else if (presetType.value === 'month') {
          // 過去月の場合: 前月1日 〜 前月末日
          const offset = presetValue.value;
          const target = new Date(now.getFullYear(), now.getMonth() + offset - 1, 1);
          const y = target.getFullYear();
          const m = target.getMonth();
          const firstDay = new Date(y, m, 1);
          const lastDay = new Date(y, m + 1, 0);
          return {
            start: formatDateKey(firstDay),
            end: formatDateKey(lastDay)
          };
        } else {
          // 任意期間: 直前日数シフト
          const { start } = activeDateRange.value;
          const d1 = createLocalDate(start);
          const days = dateRangeDaysCount.value;
          const prevEnd = addDays(d1, -1);
          const prevStart = addDays(prevEnd, -(days - 1));
          return {
            start: formatDateKey(prevStart),
            end: formatDateKey(prevEnd)
          };
        }
      });

      const prevYearPeriodRange = computed(() => {
        const now = new Date();
        if (presetType.value === 'month' && presetValue.value === 0) {
          // 今月の場合: 前年同月1日 〜 前年同月同日
          const y = now.getFullYear() - 1;
          const m = now.getMonth();
          const prevYearFirst = new Date(y, m, 1);
          const prevYearMonthLastDay = new Date(y, m + 1, 0).getDate();
          const targetDay = Math.min(now.getDate(), prevYearMonthLastDay);
          const prevYearSameDay = new Date(y, m, targetDay);
          return {
            start: formatDateKey(prevYearFirst),
            end: formatDateKey(prevYearSameDay)
          };
        } else {
          const { start, end } = activeDateRange.value;
          const d1 = createLocalDate(start);
          const d2 = createLocalDate(end);
          return {
            start: formatDateKey(addYears(d1, -1)),
            end: formatDateKey(addYears(d2, -1))
          };
        }
      });

      const activeComparisonDateRange = computed(() => {
        return comparisonMode.value === 'previous' ? previousPeriodRange.value : prevYearPeriodRange.value;
      });

      const comparisonModeLabel = computed(() => {
        if (comparisonMode.value === 'previous') {
          return (presetType.value === 'month' && presetValue.value === 0) ? '前月同日比' : '直前同期間比';
        }
        return '前年同期比';
      });

      function setDateRangeByPreset(type, val) {
        presetType.value = type;
        presetValue.value = val;
        if (type === 'year') {
          activeYearForQuarters.value = val;
        }
      }

      function isPresetActive(type, val) {
        return presetType.value === type && presetValue.value === val;
      }

      function applyCustomDateRange() {
        if (!customDateFrom.value || !customDateTo.value) {
          alert('開始日と終了日の両方を入力してください。');
          return;
        }
        if (customDateFrom.value > customDateTo.value) {
          alert('開始日は終了日以前の日付を指定してください。');
          return;
        }
        presetType.value = 'custom';
      }

      // ----------------------------------------------------
      // 4. マスタ連動 & 堅牢なマスタ判定関数
      // ----------------------------------------------------
      const currentCategoryStaffList = computed(() => {
        return allData.staffList.filter(s => s['事業区分'] === currentCategory.value);
      });

      // 稼働中スタッフのみ (有効フラグ == true)
      const activeStaffList = computed(() => {
        return currentCategoryStaffList.value.filter(s => s.isActive);
      });

      // 店舗一覧
      const storeOptions = computed(() => {
        const set = new Set();
        currentCategoryStaffList.value.forEach(s => {
          if (s['店舗名']) set.add(String(s['店舗名']).trim());
        });
        const records = currentCategory.value === '整骨院' ? allData.seikotsuinRecords : allData.daikichiRecords;
        records.forEach(r => {
          if (r['店舗名']) set.add(String(r['店舗名']).trim());
        });
        return Array.from(set).sort();
      });

      // スタッフ選択肢 (店舗連動 & 有効スタッフ優先)
      const staffOptions = computed(() => {
        let list = activeStaffList.value;
        if (selectedStore.value !== 'all') {
          list = list.filter(s => s['店舗名'] === selectedStore.value);
        }
        return list;
      });

      function onStoreChange() {
        if (selectedStaff.value !== 'all') {
          const exists = staffOptions.value.some(s => s['スタッフID'] === selectedStaff.value);
          if (!exists) selectedStaff.value = 'all';
        }
      }

      function resetFilters() {
        selectedStore.value = 'all';
        selectedStaff.value = 'all';
      }

      function switchCategory(cat) {
        if (currentCategory.value === cat) return;
        currentCategory.value = cat;
        selectedStore.value = 'all';
        selectedStaff.value = 'all';
        targetForm.businessCategory = cat;
        matrixSelectedKpi.value = cat === '整骨院' ? '次回予約_獲得率' : '提案・成約_獲得率';
        selectedReasonItem.value = cat === '整骨院' ? '次回予約' : '提案・成約';
        trendSelectedKpi.value = cat === '整骨院' ? 'nextReservation' : 'closing';
      }

      function switchViewTab(tab) {
        activeViewTab.value = tab;
        nextTick(() => {
          if (tab === 'overview') {
            updateTrendChart();
            updateReasonChart();
            updateStoreChart();
          }
        });
      }

      function isSuccessValue(category, itemDisplayName, val) {
        if (!val) return false;
        const strVal = String(val).trim();
        const matched = allData.itemMaster.find(m => 
          m['事業区分'] === category && 
          (m['項目表示名'] === itemDisplayName || m['項目名'] === itemDisplayName) && 
          (m['選択肢テキスト'] === strVal || m['選択肢名'] === strVal)
        );
        if (matched) {
          return !!matched.isSuccess;
        }
        return strVal === '獲得' || strVal === '成約' || strVal === '提案あり';
      }

      function isProposalValue(category, itemDisplayName, val) {
        if (!val) return false;
        const strVal = String(val).trim();
        const matched = allData.itemMaster.find(m => 
          m['事業区分'] === category && 
          (m['項目表示名'] === itemDisplayName || m['項目名'] === itemDisplayName) && 
          (m['選択肢テキスト'] === strVal || m['選択肢名'] === strVal)
        );
        if (matched) {
          return !matched.isUnproposedTrigger;
        }
        return strVal !== '未提案' && strVal !== '案内なし' && strVal !== '未案内' && strVal !== 'なし';
      }

      function isPriceTriggerValue(category, itemDisplayName, val) {
        if (!val) return false;
        const strVal = String(val).trim();
        const matched = allData.itemMaster.find(m => 
          m['事業区分'] === category && 
          (m['項目表示名'] === itemDisplayName || m['項目名'] === itemDisplayName) && 
          (m['選択肢テキスト'] === strVal || m['選択肢名'] === strVal)
        );
        if (matched) {
          return !!matched.isPriceTrigger;
        }
        return strVal === '成約';
      }

      // レコード内の未提案理由サマリー（獲得済み項目は除外）
      function formatReasonsSummary(rec) {
        if (!rec) return '';
        const reasons = [];
        for (const k in rec) {
          if (k.endsWith('_未提案理由') && rec[k]) {
            const itemKey = k.replace('_未提案理由', '');
            // その項目が成功ステータスの場合は未提案理由を表示しない
            if (!isSuccessValue(currentCategory.value, itemKey, rec[itemKey])) {
              reasons.push(`${itemKey}: ${rec[k]}`);
            }
          }
        }
        return reasons.join(' / ');
      }

      // ----------------------------------------------------
      // 5. 目標値ルックアップ (0は未設定として上位へフォールバック)
      // ----------------------------------------------------
      function getTargetValueForKpi(kpiName, storeName, staffId, dateRange) {
        const range = dateRange || activeDateRange.value;
        const yearMonth = normalizeYearMonth(range.start);

        // 1. スタッフレベル (staffId が指定されている場合)
        if (staffId && staffId !== 'all') {
          const staffTarget = allData.targetRecords.find(t => 
            normalizeYearMonth(t['年月']) === yearMonth &&
            t['事業区分'] === currentCategory.value &&
            t['対象レベル'] === 'スタッフ' &&
            t['対象名'] === staffId &&
            t['KPI項目'] === kpiName
          );
          if (staffTarget && typeof staffTarget.targetValue === 'number' && staffTarget.targetValue > 0) {
            let v = staffTarget.targetValue;
            if (v <= 1 && !kpiName.includes('金額') && !kpiName.includes('単価') && !kpiName.includes('総額')) v *= 100;
            return v;
          }
        }

        // 2. 店舗レベル (storeName が指定されている場合)
        const st = storeName || selectedStore.value;
        if (st && st !== 'all') {
          const storeTarget = allData.targetRecords.find(t => 
            normalizeYearMonth(t['年月']) === yearMonth &&
            t['事業区分'] === currentCategory.value &&
            t['対象レベル'] === '店舗' &&
            t['対象名'] === st &&
            t['KPI項目'] === kpiName
          );
          if (storeTarget && typeof storeTarget.targetValue === 'number' && storeTarget.targetValue > 0) {
            let v = storeTarget.targetValue;
            if (v <= 1 && !kpiName.includes('金額') && !kpiName.includes('単価') && !kpiName.includes('総額')) v *= 100;
            return v;
          }
        }

        // 3. 全社レベル
        const companyTarget = allData.targetRecords.find(t => 
          normalizeYearMonth(t['年月']) === yearMonth &&
          t['事業区分'] === currentCategory.value &&
          (t['対象レベル'] === '全社' || !t['対象レベル']) &&
          t['KPI項目'] === kpiName
        );
        if (companyTarget && typeof companyTarget.targetValue === 'number' && companyTarget.targetValue > 0) {
          let v = companyTarget.targetValue;
          if (v <= 1 && !kpiName.includes('金額') && !kpiName.includes('単価') && !kpiName.includes('総額')) v *= 100;
          return v;
        }

        return null;
      }

      function getAchievementBadgeClass(achievement) {
        if (achievement === null || achievement === undefined) {
          return 'text-slate-400 bg-slate-900 border-slate-800';
        }
        if (achievement >= 100) {
          return 'text-emerald-400 bg-emerald-950/80 border-emerald-900';
        }
        if (achievement >= 80) {
          return 'text-amber-400 bg-amber-950/80 border-amber-900';
        }
        return 'text-rose-400 bg-rose-950/80 border-rose-900';
      }

      function getAchievementTextColor(achievement) {
        if (achievement === null || achievement === undefined) return 'text-slate-500';
        if (achievement >= 100) return 'text-emerald-400 font-semibold';
        if (achievement >= 80) return 'text-amber-400 font-semibold';
        return 'text-rose-400 font-semibold';
      }

      function formatDiffText(val, unit = '') {
        if (val === null || val === undefined || isNaN(val)) return '-';
        const sign = val > 0 ? '+' : '';
        return `${sign}${val.toFixed(1)}${unit}`;
      }

      function formatDiffColor(val) {
        if (val === null || val === undefined || isNaN(val)) return 'text-slate-500';
        if (val > 0) return 'text-emerald-400';
        if (val < 0) return 'text-rose-400';
        return 'text-slate-400';
      }

      function formatPercent(val) {
        if (val === null || val === undefined || isNaN(val)) return '-';
        return `${val}%`;
      }

      // ----------------------------------------------------
      // 6. フィルタリング済みレコード（対応日基準・異動対応）
      // ----------------------------------------------------
      function filterRecordsByRange(rawList, range, storeName = 'all', staffId = 'all') {
        const { start, end } = range;
        return rawList.filter(rec => {
          // 対応日を最優先で比較
          const dateStr = extractDateStr(rec);
          if (!dateStr || dateStr < start || dateStr > end) return false;
          if (storeName !== 'all' && rec['店舗名'] !== storeName) return false;
          if (staffId !== 'all' && rec['スタッフID'] !== staffId) return false;
          return true;
        });
      }

      const filteredRecords = computed(() => {
        const raw = currentCategory.value === '整骨院' ? allData.seikotsuinRecords : allData.daikichiRecords;
        return filterRecordsByRange(raw, activeDateRange.value, selectedStore.value, selectedStaff.value);
      });

      const comparisonRecords = computed(() => {
        const raw = currentCategory.value === '整骨院' ? allData.seikotsuinRecords : allData.daikichiRecords;
        return filterRecordsByRange(raw, activeComparisonDateRange.value, selectedStore.value, selectedStaff.value);
      });

      const recentRecords = computed(() => {
        // 対応日またはタイムスタンプの降順でソート
        return [...filteredRecords.value].sort((a, b) => {
          const da = extractDateStr(a) || a['タイムスタンプ'] || '';
          const db = extractDateStr(b) || b['タイムスタンプ'] || '';
          return db.localeCompare(da);
        }).slice(0, 50);
      });

      // ----------------------------------------------------
      // 7. KPI集計ヘルパー（提案率・査定率・0件スタッフ制御対応）
      // ----------------------------------------------------
      function calculateSeikotsuinMetrics(list, storeName = 'all', staffId = 'all', range = activeDateRange.value) {
        const totalVisits = list.length;

        // 次回予約
        const nextResCount = list.filter(r => isSuccessValue('整骨院', '次回予約', r['次回予約'])).length;
        const nextResPropCount = list.filter(r => isProposalValue('整骨院', '次回予約', r['次回予約'])).length;
        const nextResRate = totalVisits > 0 ? (nextResCount / totalVisits) * 100 : 0;
        const nextResPropRate = totalVisits > 0 ? (nextResPropCount / totalVisits) * 100 : 0;
        const nextResTarget = getTargetValueForKpi('次回予約_獲得率', storeName, staffId, range);
        const nextResAchieve = nextResTarget ? Math.round((nextResRate / nextResTarget) * 100) : null;
        const nextResPropTarget = getTargetValueForKpi('次回予約_提案率', storeName, staffId, range);
        const nextResPropAchieve = nextResPropTarget ? Math.round((nextResPropRate / nextResPropTarget) * 100) : null;

        // 回数券・サブスク
        const ticketCount = list.filter(r => isSuccessValue('整骨院', '回数券・サブスク', r['回数券・サブスク'])).length;
        const ticketPropCount = list.filter(r => isProposalValue('整骨院', '回数券・サブスク', r['回数券・サブスク'])).length;
        const ticketRate = totalVisits > 0 ? (ticketCount / totalVisits) * 100 : 0;
        const ticketPropRate = totalVisits > 0 ? (ticketPropCount / totalVisits) * 100 : 0;
        const ticketTarget = getTargetValueForKpi('回数券・サブスク_獲得率', storeName, staffId, range);
        const ticketAchieve = ticketTarget ? Math.round((ticketRate / ticketTarget) * 100) : null;
        const ticketPropTarget = getTargetValueForKpi('回数券・サブスク_提案率', storeName, staffId, range);
        const ticketPropAchieve = ticketPropTarget ? Math.round((ticketPropRate / ticketPropTarget) * 100) : null;

        // 追加提案
        const addPropCount = list.filter(r => isSuccessValue('整骨院', '追加提案', r['追加提案'])).length;
        const addPropPropCount = list.filter(r => isProposalValue('整骨院', '追加提案', r['追加提案'])).length;
        const addPropRate = totalVisits > 0 ? (addPropCount / totalVisits) * 100 : 0;
        const addPropPropRate = totalVisits > 0 ? (addPropPropCount / totalVisits) * 100 : 0;
        const addPropTarget = getTargetValueForKpi('追加提案_獲得率', storeName, staffId, range);
        const addPropAchieve = addPropTarget ? Math.round((addPropRate / addPropTarget) * 100) : null;
        const addPropPropTarget = getTargetValueForKpi('追加提案_提案率', storeName, staffId, range);
        const addPropPropAchieve = addPropPropTarget ? Math.round((addPropPropRate / addPropPropTarget) * 100) : null;

        // 口コミ
        const reviewCount = list.filter(r => isSuccessValue('整骨院', '口コミ', r['口コミ'])).length;
        const reviewPropCount = list.filter(r => isProposalValue('整骨院', '口コミ', r['口コミ'])).length;
        const reviewRate = totalVisits > 0 ? (reviewCount / totalVisits) * 100 : 0;
        const reviewPropRate = totalVisits > 0 ? (reviewPropCount / totalVisits) * 100 : 0;
        const reviewTarget = getTargetValueForKpi('口コミ_獲得率', storeName, staffId, range);
        const reviewAchieve = reviewTarget ? Math.round((reviewRate / reviewTarget) * 100) : null;
        const reviewPropTarget = getTargetValueForKpi('口コミ_提案率', storeName, staffId, range);
        const reviewPropAchieve = reviewPropTarget ? Math.round((reviewPropRate / reviewPropTarget) * 100) : null;

        // 平均達成率（0件スタッフは null 除外）
        let avgAchievement = null;
        if (totalVisits > 0) {
          const achievements = [
            nextResAchieve, nextResPropAchieve,
            ticketAchieve, ticketPropAchieve,
            addPropAchieve, addPropPropAchieve,
            reviewAchieve, reviewPropAchieve
          ].filter(a => a !== null);

          if (achievements.length > 0) {
            avgAchievement = Math.round(achievements.reduce((s, a) => s + a, 0) / achievements.length);
          }
        }

        return {
          totalVisits,
          nextReservation: {
            count: nextResCount,
            rate: Math.round(nextResRate * 10) / 10,
            proposalRate: Math.round(nextResPropRate * 10) / 10,
            target: nextResTarget,
            achievement: nextResAchieve,
            proposalTarget: nextResPropTarget,
            proposalAchievement: nextResPropAchieve
          },
          ticketSubscription: {
            count: ticketCount,
            rate: Math.round(ticketRate * 10) / 10,
            proposalRate: Math.round(ticketPropRate * 10) / 10,
            target: ticketTarget,
            achievement: ticketAchieve,
            proposalTarget: ticketPropTarget,
            proposalAchievement: ticketPropAchieve
          },
          additionalProposal: {
            count: addPropCount,
            rate: Math.round(addPropRate * 10) / 10,
            proposalRate: Math.round(addPropPropRate * 10) / 10,
            target: addPropTarget,
            achievement: addPropAchieve,
            proposalTarget: addPropPropTarget,
            proposalAchievement: addPropPropAchieve
          },
          reviewAcquisition: {
            count: reviewCount,
            rate: Math.round(reviewRate * 10) / 10,
            proposalRate: Math.round(reviewPropRate * 10) / 10,
            target: reviewTarget,
            achievement: reviewAchieve,
            proposalTarget: reviewPropTarget,
            proposalAchievement: reviewPropAchieve
          },
          avgAchievement
        };
      }

      function calculateDaikichiMetrics(list, storeName = 'all', staffId = 'all', range = activeDateRange.value) {
        const totalVisits = list.length;

        // 査定実施
        const assessmentRecords = list.filter(r => r['来店・査定'] === '査定実施');
        const totalAssessments = assessmentRecords.length;
        const assessmentRate = totalVisits > 0 ? (totalAssessments / totalVisits) * 100 : 0;
        const assessmentTarget = getTargetValueForKpi('査定実施率', storeName, staffId, range);
        const assessmentAchieve = assessmentTarget ? Math.round((assessmentRate / assessmentTarget) * 100) : null;

        // 提案・成約
        const closingCount = assessmentRecords.filter(r => isSuccessValue('買取大吉', '提案・成約', r['提案・成約'])).length;
        const closingPropCount = assessmentRecords.filter(r => isProposalValue('買取大吉', '提案・成約', r['提案・成約'])).length;
        const closingRate = totalAssessments > 0 ? (closingCount / totalAssessments) * 100 : 0;
        const closingPropRate = totalAssessments > 0 ? (closingPropCount / totalAssessments) * 100 : 0;
        const closingTarget = getTargetValueForKpi('提案・成約_獲得率', storeName, staffId, range);
        const closingAchieve = closingTarget ? Math.round((closingRate / closingTarget) * 100) : null;
        const closingPropTarget = getTargetValueForKpi('提案・成約_提案率', storeName, staffId, range);
        const closingPropAchieve = closingPropTarget ? Math.round((closingPropRate / closingPropTarget) * 100) : null;

        // 金額
        let totalAmount = 0;
        assessmentRecords.forEach(r => {
          if (isPriceTriggerValue('買取大吉', '提案・成約', r['提案・成約'])) {
            totalAmount += (r.parsedPrice || 0);
          }
        });
        const avgUnitPrice = closingCount > 0 ? Math.round(totalAmount / closingCount) : 0;
        const amountTarget = getTargetValueForKpi('買取総額', storeName, staffId, range);
        const amountAchieve = (isMonthRange.value && amountTarget) ? Math.round((totalAmount / amountTarget) * 100) : null;

        const unitPriceTarget = getTargetValueForKpi('平均単価', storeName, staffId, range);
        const unitPriceAchieve = unitPriceTarget ? Math.round((avgUnitPrice / unitPriceTarget) * 100) : null;

        // 口コミ
        const reviewCount = assessmentRecords.filter(r => isSuccessValue('買取大吉', '口コミ', r['口コミ'])).length;
        const reviewPropCount = assessmentRecords.filter(r => isProposalValue('買取大吉', '口コミ', r['口コミ'])).length;
        const reviewRate = totalAssessments > 0 ? (reviewCount / totalAssessments) * 100 : 0;
        const reviewPropRate = totalAssessments > 0 ? (reviewPropCount / totalAssessments) * 100 : 0;
        const reviewTarget = getTargetValueForKpi('口コミ_獲得率', storeName, staffId, range);
        const reviewAchieve = reviewTarget ? Math.round((reviewRate / reviewTarget) * 100) : null;
        const reviewPropTarget = getTargetValueForKpi('口コミ_提案率', storeName, staffId, range);
        const reviewPropAchieve = reviewPropTarget ? Math.round((reviewPropRate / reviewPropTarget) * 100) : null;

        // 平均達成率（接客0件スタッフは null 除外）
        let avgAchievement = null;
        if (totalAssessments > 0) {
          const achievements = [
            assessmentAchieve,
            closingAchieve, closingPropAchieve,
            amountAchieve,
            unitPriceAchieve,
            reviewAchieve, reviewPropAchieve
          ].filter(a => a !== null);

          if (achievements.length > 0) {
            avgAchievement = Math.round(achievements.reduce((s, a) => s + a, 0) / achievements.length);
          }
        }

        return {
          totalVisits,
          totalAssessments,
          assessmentRate: Math.round(assessmentRate * 10) / 10,
          assessmentTarget,
          assessmentRateAchievement: assessmentAchieve,
          closingRate: {
            count: closingCount,
            rate: Math.round(closingRate * 10) / 10,
            proposalRate: Math.round(closingPropRate * 10) / 10,
            target: closingTarget,
            achievement: closingAchieve,
            proposalTarget: closingPropTarget,
            proposalAchievement: closingPropAchieve
          },
          totalAmount: {
            amount: totalAmount,
            target: amountTarget,
            achievement: amountAchieve
          },
          avgUnitPrice: {
            amount: avgUnitPrice,
            target: unitPriceTarget,
            achievement: unitPriceAchieve
          },
          reviewAcquisition: {
            count: reviewCount,
            rate: Math.round(reviewRate * 10) / 10,
            proposalRate: Math.round(reviewPropRate * 10) / 10,
            target: reviewTarget,
            achievement: reviewAchieve,
            proposalTarget: reviewPropTarget,
            proposalAchievement: reviewPropAchieve
          },
          avgAchievement
        };
      }

      function computeMetricForMatrix(records, kpi, storeName, staffId) {
        const count = records.length;
        if (count === 0) {
          return { value: 0, count: 0, achievement: null };
        }

        let val = 0;
        let target = getTargetValueForKpi(kpi, storeName, staffId);
        const isMoney = kpi === '買取総額' || kpi === '平均単価';

        if (currentCategory.value === '整骨院') {
          if (kpi === '次回予約_獲得率') {
            val = (records.filter(r => isSuccessValue('整骨院', '次回予約', r['次回予約'])).length / count) * 100;
          } else if (kpi === '次回予約_提案率') {
            val = (records.filter(r => isProposalValue('整骨院', '次回予約', r['次回予約'])).length / count) * 100;
          } else if (kpi === '回数券・サブスク_獲得率') {
            val = (records.filter(r => isSuccessValue('整骨院', '回数券・サブスク', r['回数券・サブスク'])).length / count) * 100;
          } else if (kpi === '回数券・サブスク_提案率') {
            val = (records.filter(r => isProposalValue('整骨院', '回数券・サブスク', r['回数券・サブスク'])).length / count) * 100;
          } else if (kpi === '追加提案_獲得率') {
            val = (records.filter(r => isSuccessValue('整骨院', '追加提案', r['追加提案'])).length / count) * 100;
          } else if (kpi === '追加提案_提案率') {
            val = (records.filter(r => isProposalValue('整骨院', '追加提案', r['追加提案'])).length / count) * 100;
          } else if (kpi === '口コミ_獲得率') {
            val = (records.filter(r => isSuccessValue('整骨院', '口コミ', r['口コミ'])).length / count) * 100;
          } else if (kpi === '口コミ_提案率') {
            val = (records.filter(r => isProposalValue('整骨院', '口コミ', r['口コミ'])).length / count) * 100;
          }
        } else {
          const assess = records.filter(r => r['来店・査定'] === '査定実施');
          const assessCount = assess.length;

          if (kpi === '査定実施率') {
            val = (assessCount / count) * 100;
          } else if (kpi === '提案・成約_獲得率') {
            val = assessCount > 0 ? (assess.filter(r => isSuccessValue('買取大吉', '提案・成約', r['提案・成約'])).length / assessCount) * 100 : 0;
          } else if (kpi === '提案・成約_提案率') {
            val = assessCount > 0 ? (assess.filter(r => isProposalValue('買取大吉', '提案・成約', r['提案・成約'])).length / assessCount) * 100 : 0;
          } else if (kpi === '買取総額') {
            let sum = 0;
            assess.forEach(r => {
              if (isPriceTriggerValue('買取大吉', '提案・成約', r['提案・成約'])) sum += (r.parsedPrice || 0);
            });
            val = sum;
          } else if (kpi === '平均単価') {
            const closing = assess.filter(r => isSuccessValue('買取大吉', '提案・成約', r['提案・成約'])).length;
            let sum = 0;
            assess.forEach(r => {
              if (isPriceTriggerValue('買取大吉', '提案・成約', r['提案・成約'])) sum += (r.parsedPrice || 0);
            });
            val = closing > 0 ? Math.round(sum / closing) : 0;
          } else if (kpi === '口コミ_獲得率') {
            val = assessCount > 0 ? (assess.filter(r => isSuccessValue('買取大吉', '口コミ', r['口コミ'])).length / assessCount) * 100 : 0;
          } else if (kpi === '口コミ_提案率') {
            val = assessCount > 0 ? (assess.filter(r => isProposalValue('買取大吉', '口コミ', r['口コミ'])).length / assessCount) * 100 : 0;
          }
        }

        val = isMoney ? val : Math.round(val * 10) / 10;
        // 金額KPIは達成率計算を除外（ヒートマップ色分けしない）
        const achieve = (!isMoney && target) ? Math.round((val / target) * 100) : null;

        return {
          value: val,
          count: count,
          achievement: achieve
        };
      }

      function exportFilteredToCsv() {
        const records = filteredRecords.value;
        if (!records || records.length === 0) return;

        const excludeKeys = new Set(['送信ID', 'parsedPrice', 'isActive', 'isSuccess', 'isUnproposedTrigger', 'isPriceTrigger']);
        const headers = Object.keys(records[0]).filter(k => !excludeKeys.has(k));

        let csv = '\uFEFF' + headers.join(',') + '\n';
        records.forEach(r => {
          const row = headers.map(h => {
            let v = r[h] !== undefined && r[h] !== null ? String(r[h]) : '';
            // 数式インジェクション対策 (=, +, -, @ で始まる場合は先頭にシングルクォート付与)
            if (/^[=+\-@]/.test(v)) {
              v = "'" + v;
            }
            if (v.includes(',') || v.includes('\n') || v.includes('"')) {
              v = '"' + v.replace(/"/g, '""') + '"';
            }
            return v;
          });
          csv += row.join(',') + '\n';
        });

        const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
        const url = URL.createObjectURL(blob);
        const link = document.createElement('a');
        link.href = url;
        link.download = `接客実績_${currentCategory.value}_${activeDateRange.value.start}_${activeDateRange.value.end}.csv`;
        link.click();
        URL.revokeObjectURL(url);
      }
